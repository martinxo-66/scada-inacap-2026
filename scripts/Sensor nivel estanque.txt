Sensor nivel estanque

cat << 'EOF' > /sensor_n.py
import time
import urllib.request
import json

while True:
    try:
        # Consulta física al estanque correcto (192.168.1.10)
        with urllib.request.urlopen("http://192.168.1.10:5001", timeout=3) as resp:
            data = json.loads(resp.read().decode('utf-8'))

        # Enviar al PLC Central (192.168.1.20)
        req = urllib.request.Request(
            "http://192.168.1.20:5000",
            data=json.dumps(data).encode('utf-8'),
            headers={'Content-Type': 'application/json'}
        )
        urllib.request.urlopen(req, timeout=3)
        print("[SENSOR] Telemetría enviada con éxito al PLC:", data)
    except Exception as e:
        print("Error en sensor:", e)
    time.sleep(3)
EOF
