import time
import urllib.request
import json

while True:
    try:
        with urllib.request.urlopen("http://192.168.1.20:5000", timeout=3) as resp:
            data = json.loads(resp.read().decode('utf-8'))
        
        print(f"[PUENTE-AWS] Sincronizando telemetría OT con la nube -> Nivel: {data.get('nivel_estanque')}% | Cloro: {data.get('cloro_ppm')}ppm | Alarma: {data.get('alarma')}")
    except Exception as e:
        print("[PUENTE-AWS] Error de enlace con red local OT:", e)
    
    time.sleep(5)
