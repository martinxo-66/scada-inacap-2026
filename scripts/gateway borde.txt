cat << 'EOF' > /gateway.py
import time, urllib.request, socket, json

while True:
    try:
        req = urllib.request.Request("http://192.168.10.20:5000", headers={"User-Agent": "Mozilla/5.0"})
        with urllib.request.urlopen(req, timeout=3) as r:
            payload = r.read().decode()
            print("[GATEWAY] Polling al PLC exitoso. Enviando a MQTT...")
            s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
            s.connect(('192.168.10.18', 1883))
            s.sendall(payload.encode())
            s.close()
    except Exception as e:
        print("[GATEWAY] Error:", e)
    time.sleep(5)
EOF
