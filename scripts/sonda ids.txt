import time
import urllib.request
import json

print("[SONDA-IDS] Sistema de Detección de Intrusos OT activo y monitoreando tráfico...")

while True:
    try:
        with urllib.request.urlopen("http://192.168.1.20:5000", timeout=3) as resp:
            data = json.loads(resp.read().decode('utf-8'))
            
        alarma = data.get("alarma", "NORMAL")
        if alarma == "CRITICA":
            print("[ALERTA-IDS] ¡Anomalía detectada en procesos críticos! Valores fuera de rango seguro.")
        else:
            print("[SONDA-IDS] Tráfico y parámetros industriales dentro del umbral seguro. Sin amenazas.")
    except Exception as e:
        print("[SONDA-IDS] Advertencia: Pérdida temporal de visibilidad en el bus de red:", e)
        
    time.sleep(4)
