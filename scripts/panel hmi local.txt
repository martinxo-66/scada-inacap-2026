import urllib.request
import json
from http.server import HTTPServer, BaseHTTPRequestHandler

class HMIHandler(BaseHTTPRequestHandler):
    def do_GET(self):
        try:
            req = urllib.request.Request("http://192.168.1.20:5000")
            with urllib.request.urlopen(req, timeout=3) as resp:
                data = json.loads(resp.read().decode('utf-8'))
            
            html = f"""
            <!DOCTYPE html>
            <html>
            <head>
                <title>HMI - Planta Tratamiento de Agua INACAP</title>
                <meta http-equiv="refresh" content="3">
                <style>
                    body {{ font-family: Arial, sans-serif; background: #121212; color: #e0e0e0; text-align: center; padding: 50px; }}
                    .card {{ background: #1e1e1e; padding: 20px; border-radius: 10px; display: inline-block; box-shadow: 0 4px 8px rgba(0,0,0,0.3); }}
                    h2 {{ color: #03dac6; }}
                    .normal {{ color: #4caf50; font-weight: bold; }}
                    .critica {{ color: #f44336; font-weight: bold; }}
                </style>
            </head>
            <body>
                <div class="card">
                    <h2>Panel HMI - Supervisión SCADA</h2>
                    <p><b>Nivel de Estanque:</b> {data.get('nivel_estanque')}%</p>
                    <p><b>Nivel de Cloro:</b> {data.get('cloro_ppm')} ppm</p>
                    <p><b>Estado de Bomba:</b> {data.get('bomba')}</p>
                    <p><b>Estado de Alarma:</b> <span class="{data.get('alarma').lower()}">{data.get('alarma')}</span></p>
                </div>
            </body>
            </html>
            """
            self.send_response(200)
            self.send_header('Content-Type', 'text/html; charset=utf-8')
            self.end_headers()
            self.wfile.write(html.encode('utf-8'))
        except Exception as e:
            self.send_response(500)
            self.end_headers()
            self.wfile.write(f"Error conectando con PLC Central: {e}".encode())

    def log_message(self, format, *args):
        return

if __name__ == '__main__':
    PORT = 8080
    print(f"[PANEL-HMI] Servidor web HMI activo en puerto {PORT}...")
    HTTPServer(("0.0.0.0", PORT), HMIHandler).serve_forever()
EOF
